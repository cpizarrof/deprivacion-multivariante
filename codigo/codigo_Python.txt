# Grafico de estrellas

def star_plot(X, n_rows=2, n_cols=4, scale=True, title="Star Plot"):
    import numpy as np
    import matplotlib.pyplot as plt

    if scale:
        X = (X - X.mean()) / X.std()

    labels = X.columns
    num_vars = len(labels)

    angles = np.linspace(0, 2*np.pi, num_vars, endpoint=False).tolist()
    angles += angles[:1]

    fig, axes = plt.subplots(n_rows, n_cols, subplot_kw=dict(polar=True))
    axes = axes.flatten()

    for i, ax in enumerate(axes):
        if i >= len(X):
            ax.axis("off")
            continue

        values = X.iloc[i].tolist()
        values += values[:1]

        color = plt.cm.rainbow(i / len(X))

        ax.plot(angles, values, color=color)
        ax.fill(angles, values, color=color, alpha=0.2)

        ax.set_title(X.index[i], size=9)
        ax.set_xticks(angles[:-1])
        ax.set_xticklabels(labels, fontsize=6)
        ax.set_yticks([])

    plt.suptitle(title)
    plt.tight_layout()
    plt.show()

from matplotlib.patches import Ellipse
def elipse(ax, mu, S, t, **kw):
    ev, V = np.linalg.eigh(S)
    ang = np.degrees(np.arctan2(V[1,1], V[0,1]))
    ax.add_patch(Ellipse(mu, 2*t*np.sqrt(ev[1]), 2*t*np.sqrt(ev[0]), angle=ang,
                         fill=False, **kw))

def descomp(y, g):
    vt = ((y - y.mean())**2).mean()
    va = pd.Series(y).groupby(g.to_numpy()).apply(lambda v: ((v-v.mean())**2).sum()).sum()/len(y)
    return round(100*va/vt, 1), round(100*(1-va/vt), 1)

from sklearn.metrics import silhouette_score, silhouette_samples
print(silhouette_score(Z, grupos))

M = squareform(D)
def dunn(g):
    inter, intra = np.inf, 0.0
    for a in np.unique(g):
        ia = np.where(g == a)[0]
        intra = max(intra, M[np.ix_(ia, ia)].max())
        for b in np.unique(g):
            if b > a:
                ib = np.where(g == b)[0]
                inter = min(inter, M[np.ix_(ia, ib)].min())
    return inter/intra
print(dunn(grupos))

sil, dun = [], []
for k in range(2, 11):
    g = fcluster(hc, k, criterion="maxclust")
    sil.append(silhouette_score(Z, g))
