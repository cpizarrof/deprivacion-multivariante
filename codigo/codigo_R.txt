#descomposicion de la varianza de la CP1: cuanto es de adentro y cuanto de afuera
#es la V_T = V_a + V_e de la Clase 1, aplicada a region, provincia y comuna
descomp=function(y,g){
  vt=mean((y-mean(y))^2)
  va=sum(tapply(y,g,function(v) mean((v-mean(v))^2)*length(v)))/length(y)
  c(intra=100*va/vt, inter=100*(1-va/vt))
}
for(k in 1:3){
  cat("CP",k,"\n")
  print(rbind(Region=descomp(acp$x[,k],bd$codigo_region),
              Provincia=descomp(acp$x[,k],bd$codigo_provincia),
              Comuna=descomp(acp$x[,k],bd$codigo_comuna)))
}

folds=sample(rep_len(1:10,n))
prob=rep(0,n)
for(k in 1:10){
  m=glm(y~.,data=datos[folds!=k,],family="binomial")
  prob[folds==k]=predict(m,newdata=datos[folds==k,],type="response")
}

require(ROCR)
pred=prediction(prob,datos$y)
perf=performance(pred,"tpr","fpr")
plot(perf,lwd=2)
abline(0,1,lty=2)
auc=performance(pred,"auc")@y.values[[1]]
auc

require(pROC)
objeto_roc=roc(datos$y,prob)
valores_optimos=coords(objeto_roc,x="best",best.method="youden")
valores_optimos    #umbral, especificidad y sensibilidad

#la matriz de confusion en los dos cortes que van en la lamina
corte=function(u){
  t0=table(datos$y,factor(ifelse(prob>u,1,0),levels=c(0,1)))
  cat("umbral",round(u,4),
      " sens",round(100*t0[2,2]/sum(t0[2,]),1),
      " esp",round(100*t0[1,1]/sum(t0[1,]),1),
      " acierto",round(100*sum(diag(t0))/sum(t0),1),"\n")
  print(t0)
}
corte(valores_optimos$threshold)   #corte de Youden: maximiza sens+esp-1
corte(0.5)                          #corte estricto: prioriza evitar falsas alarmas
